use neo_contract::prelude::*;

/// Production-ready NEP-17 token with comprehensive features
#[contract_author("Neo Rust Team")]
#[contract_version("2.0.0")]
#[contract_standards("NEP-17")]
#[contract_permission("*:*")]
#[contract_meta("Website", "https://neo-contract-rs.netlify.app")]
#[contract]
pub struct AdvancedToken {
    // Core token data
    total_supply: StorageItem,
    balances: StorageMap,

    // Administrative controls
    owner: StorageItem,
    paused: StorageItem,
}

#[contract_impl]
impl AdvancedToken {
    #[method]
    #[safe]
    pub fn symbol(&self) -> ByteString {
        ByteString::from("NEORUST")
    }

    #[method]
    #[safe]
    pub fn decimals(&self) -> u8 { 8 }

    #[method]
    #[safe]
    pub fn total_supply(&self) -> Int256 {
        self.total_supply.get().unwrap_or_default()
    }

    #[method]
    pub fn transfer(&self, from: H160, to: H160, amount: Int256, data: Option) -> bool {
        // Authorization check
        if !Runtime::check_witness(from) {
            return false;
        }

        // Pause check
        if self.paused.get().unwrap_or(false) {
            return false;
        }

        // Balance validation and transfer
        let from_balance = self.balances.get(from).unwrap_or_default();
        if from_balance < amount {
            return false;
        }

        // Execute transfer
        self.balances.put(from, from_balance - amount);
        let to_balance = self.balances.get(to).unwrap_or_default();
        self.balances.put(to, to_balance + amount);

        // Emit transfer event
        Runtime::notify("Transfer", &[from.into(), to.into(), amount.into()]);

        // Post-transfer hook for contracts
        if let Some(contract_data) = data {
            self.on_nep17_payment(from, amount, contract_data);
        }

        true
    }

    #[method]
    pub fn mint(&self, to: H160, amount: Int256) -> bool {
        // Only owner can mint
        if !Runtime::check_witness(self.owner.get().unwrap()) {
            return false;
        }

        let supply = self.total_supply.get().unwrap_or_default();
        self.total_supply.put(supply + amount);

        let balance = self.balances.get(to).unwrap_or_default();
        self.balances.put(to, balance + amount);

        Runtime::notify("Transfer", &[H160::zero().into(), to.into(), amount.into()]);
        true
    }
}

Memory Safety

✓ Rust: Compile-time memory safety prevents buffer overflows, null pointer dereferences, and memory leaks
✗ C#/Other: Runtime memory management with potential for memory-related vulnerabilities

Performance

✓ Rust: Zero-cost abstractions and aggressive optimizations result in smaller, faster contracts
✗ C#/Other: Runtime overhead and larger bytecode size increase gas costs

Type System

✓ Rust: Powerful type system catches logic errors at compile time, preventing runtime failures
✗ C#/Other: Many errors only discovered at runtime, potentially after deployment

Developer Experience

✓ Rust: Modern tooling with Cargo, excellent IDE support, comprehensive package ecosystem
✗ C#/Other: Limited blockchain-specific tooling and smaller ecosystem for smart contracts

Compilation Pipeline

✓ Rust: Direct WASM compilation with custom neo-wasm translator for optimal Neo VM bytecode
✗ C#/Other: Multiple compilation steps with potential optimization losses

Community & Future

✓ Rust: Rapidly growing blockchain adoption, WebAssembly standard, cross-platform compatibility
✗ C#/Other: Platform-specific limitations and smaller blockchain developer community

Getting Started

Follow these simple steps to start building Neo N3 smart contracts with Rust.

1

Install Prerequisites

Make sure you have Rust and the WebAssembly target installed.

$ rustup target add wasm32-unknown-unknown
2

Create a New Project

Set up a new Rust project for your Neo smart contract.

$ cargo new --lib my-neo-contract
$ cd my-neo-contract
3

Configure Your Project

Add neo-contract-rs as a dependency in your Cargo.toml.

[dependencies]
neo-contract = { version = "0.1.0" }

[lib]
crate-type = ["cdylib"]
4

Write Your Contract

Create your smart contract using the neo-contract-rs framework.

use neo_contract::prelude::*;

#[contract]
pub struct MyContract {
    // Contract state
    message: StorageItem,
}

#[contract_impl]
impl MyContract {
    pub fn init() -> Self {
        let ctx = Storage::get_context();
        Self {
            message: StorageItem::new(ctx, b"message"),
        }
    }

    #[method]
    #[safe]
    pub fn hello(&self) -> ByteString {
        self.message.get().unwrap_or_else(|| ByteString::from("Hello, Neo!"))
    }

    #[method]
    pub fn set_message(&self, new_message: ByteString) -> bool {
        self.message.put(new_message);
        true
    }
}
5

Build with Makefile

Use the provided Makefile for automated building and compilation.

$ make build    # Build WASM
$ make compile  # Convert to NEF with neo-wasm
$ make all      # Build, compile, and optimize
6

Deploy and Test

Your contract is now ready for deployment to Neo N3 networks.

$ ls build/
my_contract.nef      # Neo Executable Format
my_contract.manifest.json  # Contract manifest

02 - Simple Storage

Storage operations and patterns using StorageMap for key-value data persistence.

use neo_contract::prelude::*;

#[contract]
pub struct SimpleStorage {
    data: StorageMap,
    counter: StorageItem,
}

#[contract_impl]
impl SimpleStorage {
    #[method]
    pub fn put(&self, key: ByteString, value: ByteString) -> bool {
        self.data.put(key.clone(), value);

        // Increment counter
        let count = self.counter.get().unwrap_or(0);
        self.counter.put(count + 1);

        Runtime::notify("DataStored", &[key.into(), count.into()]);
        true
    }

    #[method]
    #[safe]
    pub fn get(&self, key: ByteString) -> Option {
        self.data.get(key)
    }
}

05 - NEP-11 NFT

Non-fungible token with metadata support, enumeration, and administrative controls.

use neo_contract::prelude::*;

#[contract_standards("NEP-11")]
#[contract]
pub struct NEP11NFT {
    tokens: StorageMap,
    token_metadata: StorageMap,
    total_supply: StorageItem,
    owner: StorageItem,
}

#[contract_impl]
impl NEP11NFT {
    #[method]
    pub fn mint(&self, to: H160, token_id: ByteString, metadata: ByteString) -> bool {
        if !Runtime::check_witness(self.owner.get().unwrap()) {
            return false;
        }

        if self.tokens.get(token_id.clone()).is_some() {
            return false; // Token already exists
        }

        self.tokens.put(token_id.clone(), to);
        self.token_metadata.put(token_id.clone(), metadata);

        let supply = self.total_supply.get().unwrap_or_default();
        self.total_supply.put(supply + 1);

        Runtime::notify("Transfer", &[H160::zero().into(), to.into(), 1.into(), token_id.into()]);
        true
    }
}

09 - Simple DEX

Decentralized exchange with automated market maker, liquidity pools, and token swaps.

use neo_contract::prelude::*;

#[contract]
pub struct SimpleDEX {
    pools: StorageMap<(H160, H160), LiquidityPool>,
    user_liquidity: StorageMap<(H160, H160, H160), Int256>,
    fee_rate: StorageItem, // Basis points (e.g., 30 = 0.3%)
}

#[contract_impl]
impl SimpleDEX {
    #[method]
    pub fn swap(&self, token_in: H160, token_out: H160, amount_in: Int256, min_amount_out: Int256) -> bool {
        let caller = Runtime::get_calling_script_hash();

        let pool_key = self.get_pool_key(token_in, token_out);
        let mut pool = self.pools.get(pool_key).unwrap();

        let amount_out = self.calculate_swap_amount(
            amount_in, pool.reserve_a, pool.reserve_b
        );

        if amount_out < min_amount_out {
            return false;
        }

        // Execute swap logic...
        pool.reserve_a += amount_in;
        pool.reserve_b -= amount_out;
        self.pools.put(pool_key, pool);

        Runtime::notify("Swap", &[caller.into(), token_in.into(), token_out.into(), amount_in.into(), amount_out.into()]);
        true
    }
}

13 - NFT Marketplace

Complete marketplace with modular architecture, listings, auctions, offers, and royalty distribution.

use neo_contract::prelude::*;

#[contract]
pub struct NFTMarketplace {
    listings: StorageMap,
    auctions: StorageMap,
    offers: StorageMap,
    marketplace_fee: StorageItem,
    next_listing_id: StorageItem,
}

#[contract_impl]
impl NFTMarketplace {
    #[method]
    pub fn create_listing(&self, nft_contract: H160, token_id: ByteString, price: Int256, currency: H160) -> u64 {
        let seller = Runtime::get_calling_script_hash();
        let listing_id = self.next_listing_id.get().unwrap_or(1);

        let listing = Listing {
            seller,
            nft_contract,
            token_id: token_id.clone(),
            price,
            currency,
            active: true,
            created_at: Runtime::get_time(),
        };

        self.listings.put(listing_id, listing);
        self.next_listing_id.put(listing_id + 1);

        Runtime::notify("ListingCreated", &[listing_id.into(), seller.into(), nft_contract.into(), token_id.into(), price.into()]);
        listing_id
    }
}
