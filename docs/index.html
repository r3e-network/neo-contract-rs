Simple and Intuitive API

#[neo_contract::contract]
#[neo_contract::contract_author("Your Name")]
#[neo_contract::contract_description("A simple token contract")]
#[neo_contract::contract_version("0.1.0")]
mod my_contract {
#[neo(storage)]
pub struct MyContract {
total_supply: Int256,
balances: Map<H160, Int256>,
}

impl MyContract {
#[neo(constructor)]
pub fn new(initial_supply: Int256) -> Self {
// Implementation...
Self { total_supply: initial_supply, balances: Map::new() }
}

#[neo(message)]
pub fn transfer(&mut self, from: H160, to: H160, amount: Int256) -> bool {
// Implementation...
true
}
}
}
